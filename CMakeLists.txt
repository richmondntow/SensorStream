cmake_minimum_required(VERSION 3.16)
project(SensorStream LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(sensorstream_server src/server.cpp)
target_link_libraries(sensorstream_server PRIVATE Threads::Threads)
target_compile_options(sensorstream_server PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:sensorstream_server>)
